% B1: falha no meio da importação. Conferido no código da 3.0.0: runner.py (etapas 1 e 2,
% cabeçalhos e _require_dbms antes de ler as linhas de dados); stream_runner.py (leituras de
% inferência e de validação antes da primeira gravação; depois, um SGBD por vez, na ordem de
% validation.DBMS_NAMES, sem desfazer; uma falha na gravação termina com código 1 e a mensagem
% "already written: ..."); sinks: ON CONFLICT DO NOTHING com chave no PostgreSQL, INSERT do
% Cassandra (upsert), SET do Redis, MERGE do Neo4j, insert_many do MongoDB (sem _id derivado
% da chave; com um _id mapeado, a reexecução para na chave repetida). Coerente com o slide 23.
\begin{frame}{Falha no meio da importação}
  \small
  \begin{itemize}
    \setlength{\itemsep}{5pt}
    \item \destaque{Antes de gravar}, as três etapas de validação conferem configuração e
          dados, e todos os SGBDs destino precisam responder. Qualquer falha: nada é gravado.
    \item \destaque{Um SGBD por vez}, em lotes: PostgreSQL, MongoDB, Cassandra, Redis, Neo4j.
    \item \destaque{Sem atomicidade entre SGBDs} (em aberto, slide 23): não há transação
          comum nem desfazer (\textit{rollback}).
    \item \destaque{Falha durante a gravação}, como a queda de um SGBD: os anteriores ficam
          gravados, o atual fica parcial, os seguintes ficam vazios; a saída lista os já
          gravados.
    \item \destaque{Reexecução} com \texttt{-{}-only}: com chave, PostgreSQL, Cassandra, Redis
          e Neo4j não duplicam. No MongoDB, sem \texttt{\_id} mapeado, duplica documentos;
          com \texttt{\_id}, para na chave repetida.
  \end{itemize}
\end{frame}

% B2: coerção de tipos. Conferido no código da 3.0.0: csv_reader (tudo lido como texto; a
% leitura de inferência agrega cada coluna sobre o arquivo inteiro, em blocos; ordem:
% booleano, número, data e hora acima de 85% dos não vazios, texto; vazia sem valor; um
% valor só de dígitos nunca conta como data, para que uma coluna de IDs não vire data);
% casting.cast_frame (vazio -> None nos tipos convertidos; colunas de texto mantêm o valor
% bruto, e o vazio fica ""); leitura de validação (relatório §4.2.4: valor que não converte,
% valor que o db_type recusaria, chave vazia e referência pendente são apontados, com
% contagem e exemplos, e nada é gravado). Futuro: "políticas de coerção configuráveis"
% (Cap. 6).
% Minipage [c]: as setas ficam na altura do meio dos painéis.
\newcommand{\bktipo}[2]{% #1 tipo, #2 regra
  \begin{minipage}[c]{0.215\textwidth}\begin{painel}{#1}{}\raggedright #2\end{painel}%
  \end{minipage}}
\newcommand{\bkseta}{\hfill{\color{ufscazul}$\rightarrow$}\hfill}
\begin{frame}{Coerção de tipos}
  \small O CSV é lido como \destaque{texto}. A \destaque{leitura de inferência} percorre o
  arquivo inteiro e classifica cada coluna, nesta ordem:\par
  \medskip
  \begingroup\footnotesize\tcbset{equal height group=b2}%
  \bktipo{1. Booleano}{todos os valores não vazios são \texttt{true} ou \texttt{false}}\bkseta
  \bktipo{2. Número}{todos são inteiros ou reais}\bkseta
  \bktipo{3. Data e hora}{mais de 85\% dos não vazios são datas}\bkseta
  \bktipo{4. Texto}{nenhum dos casos anteriores}\par
  \endgroup
  \medskip
  \footnotesize
  \begin{itemize}
    \setlength{\itemsep}{2pt}
    \item Célula vazia vira nulo em número, booleano e data; em texto, fica vazia.
    \item Valor só com dígitos nunca conta como data: uma coluna de IDs não vira data.
    \item A \destaque{leitura de validação} aponta, com arquivo, linha, coluna e valor, o
          valor que não converte para o tipo da coluna e o que o tipo declarado
          (\texttt{db\_type}) recusaria. Nada é gravado.
    \item No PostgreSQL, chave repetida não duplica; chave vazia é apontada na validação.
  \end{itemize}
  \vfill
  \bknota{Política de coerção configurável: trabalho futuro (Cap.~6 do relatório).}
\end{frame}

% B3: por que configurar à mão? (relatório §1.1: mapeamento declarativo em JSON, que permite
% a validação prévia; §2.1: a escolha do SGBD depende de como se quer manipular os dados;
% §4.2: mapeamento automático, tipos inferidos; §3.3: carga inicial e reprodutível).
% O uso de LLM não existe na ferramenta: o slide o trata como caminho possível, fora do
% escopo do trabalho.
\begin{frame}{Por que configurar à mão?}
  \bkpar{b3}{%
    \begin{painel}{Mapeamento declarativo}{o que a ferramenta faz hoje}
      \begin{bklista}
        \item Quem projeta decide que dado vai para cada SGBD, com chaves, partições e
              arestas.
        \item O arquivo pode ser revisto, e o \texttt{-{}-dry-run} o valida sem gravar.
        \item A mesma configuração reproduz a mesma carga.
        \item Parte já é automática: sem \texttt{columns}, entram todas as colunas, com
              tipos inferidos.
      \end{bklista}
    \end{painel}}{%
    \begin{painel}{Mapeamento inferido}{por um modelo de linguagem (LLM)}
      \begin{bklista}
        \item Poupa a escrita inicial do arquivo.
        \item O CSV não diz como a aplicação consulta os dados. É isso que decide o SGBD de
              cada dado.
        \item A proposta ainda precisa de revisão.
      \end{bklista}
    \end{painel}}
  \smallskip
  \footnotesize\destaque{Complementares:} um LLM poderia propor o arquivo de importação, e o
  \texttt{-{}-dry-run} o conferiria sem gravar. Caminho possível, fora do escopo deste
  trabalho.\par
\end{frame}

% B6: ameaças à validade (relatório v2.4 §5.4.1, sec:ameacas-validade; dados sintéticos:
% §5.1.1; precisão: §5.1.2).
\begin{frame}{Ameaças à validade}
  \begingroup\footnotesize\tcbset{equal height group=b6, halign=flush left}%
  \begin{minipage}[t]{0.32\textwidth}\vspace{0pt}
    \begin{painel}{De construção}{mede o que se quer?}
      \begin{bklista}
        \item Tempo: 2,7\% entre campanhas, numa só célula (históricas: cerca de 20\%).
        \item Memória estável: menos de 0,2\%.
        \item Tempo e memória de campanhas separadas; cargas diferentes nas históricas.
      \end{bklista}
    \end{painel}
  \end{minipage}\hfill
  \begin{minipage}[t]{0.32\textwidth}\vspace{0pt}
    \begin{painel}{Interna}{o efeito vem do fator?}
      \begin{bklista}
        \item Gerador determinístico e SGBDs limpos antes de cada repetição.
        \item Volume moderado: até \num{100000} linhas; o bloco de 256~MB já lê cada fonte
              inteira.
        \item Sem caches quentes nem execuções concorrentes.
      \end{bklista}
    \end{painel}
  \end{minipage}\hfill
  \begin{minipage}[t]{0.32\textwidth}\vspace{0pt}
    \begin{painel}{Externa}{vale fora deste ambiente?}
      \begin{bklista}
        \item Uma máquina, Windows, cinco SGBDs em contêineres com 8~GB: não é produção.
        \item Cassandra limitado pelo \textit{heap} de 1~GB.
        \item Dados sintéticos de um só domínio, o e-commerce (§5.1.1).
      \end{bklista}
    \end{painel}
  \end{minipage}\par
  \endgroup
  \vfill
  \bknota{Fonte: §5.4.1 do relatório.}
\end{frame}
