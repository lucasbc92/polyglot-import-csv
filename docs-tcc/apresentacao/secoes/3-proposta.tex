% Slide 15: estratégia de escrita e modelo de execução (relatório v2.4 §4.5, §4.5.1, §4.5.2)
% Só conceitos: os números ficam para a avaliação. Vocabulário do relatório: ingênua
% (naive) x em lote (batch); materialização x leitura em blocos (streaming). A ingênua e a
% materialização não existem na 3.0.0: são as linhas de base históricas de QP4 a QP6.
% Padrões e faixas de cli.py: --chunk-size 8 MB (8 a 256), --batch-size 1024 (1024 a 8192).
\newcommand{\fator}[3]{% #1 grupo de altura, #2 painel da esquerda, #3 painel da direita
  \tcbset{equal height group=#1}%
  \begin{minipage}[c]{0.455\textwidth}\footnotesize #2\end{minipage}\hfill
  \begin{minipage}[c]{0.07\textwidth}\centering\Large\color{ufscazul}$\times$%
  \end{minipage}\hfill
  \begin{minipage}[c]{0.455\textwidth}\footnotesize #3\end{minipage}\par}
\begin{frame}{Escrita e leitura em blocos}
  \begingroup
  \footnotesize\destaque{Estratégia de escrita}\enspace
  {\scriptsize\color{black!60} como os valores viram tipos nativos e são gravados}\par
  \smallskip
  \fator{escrita}{%
    \begin{painel}{ingênua (\textit{naive})}{TCC1, linha de base histórica}
      Convertia cada célula e gravava cada linha com um comando próprio.
    \end{painel}}{%
    \begin{painel}{em lote (\textit{batch})}{a única da versão final}
      Converte uma coluna inteira de uma vez e grava muitos registros por comando.
    \end{painel}}
  \onslide<2->{%
  \medskip
  \footnotesize\destaque{Modelo de execução}\enspace
  {\scriptsize\color{black!60} como os arquivos ocupam a memória}\par
  \smallskip
  \fator{execucao}{%
    \begin{painel}{materialização}{linha de base histórica}
      Carregava cada fonte inteira na memória: um arquivo maior que ela não chegaria a
      ser carregado.
    \end{painel}}{%
    \begin{painel}{leitura em blocos (\textit{streaming})}{o único da versão final}
      Lê em blocos de \texttt{-{}-chunk-size} (8~MB) e grava em lotes de
      \texttt{-{}-batch-size} (1024 linhas). O pico acompanha o bloco.
    \end{painel}}}
  \onslide<3->{%
  \medskip
  \footnotesize As linhas de base históricas são as que a avaliação compara
  (QP4 a QP6).\par}
  \endgroup
\end{frame}
