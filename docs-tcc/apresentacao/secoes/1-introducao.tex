\begin{frame}{Persistência poliglota e o problema}
  \begin{itemize}
    \item \destaque{Persistência poliglota}: cada tipo de dado fica no SGBD (sistema
          gerenciador de banco de dados) mais adequado.
    \item Cada SGBD tem o seu \destaque{importador nativo} de CSV, que atende só a ele.
    \item<2-> \destaque{Problema:} nenhuma ferramenta encontrada importa o mesmo CSV nos
          cinco.
  \end{itemize}
  \vfill
  \centering
  \begin{tikzpicture}[x=1mm, y=1mm, font=\footnotesize,
      caixa/.style={draw=ufscborda, fill=white, rounded corners=0.6mm, inner sep=2pt,
                    minimum height=5.6mm, anchor=west, align=left, text=ufsctexto},
      seta/.style={->, >=stealth, draw=ufscazul!70, thick}]
    \node[caixa, fill=ufscfundo, align=center, minimum width=28mm] (csv) at (0,0)
      {arquivos CSV\\[-1pt]{\scriptsize e-commerce}};
    \node[caixa, fill=ufscazul, draw=ufscazul, text=white, align=center, minimum width=36mm]
      (tool) at (34,0) {\bfseries Polyglot Import CSV\\[-1pt]{\scriptsize configuração JSON}};
    \foreach \sgbd/\tipo/\uso [count=\i] in {
        PostgreSQL/relacional/pedidos e inventário,
        MongoDB/documento/catálogo de produtos,
        Redis/chave-valor/carrinho e sessão,
        Cassandra/colunar/log de atividade,
        Neo4j/grafo/recomendações} {
      \node[caixa, minimum width=66mm] (n\i) at (77, {18.6-6.2*\i})
        {\textbf{\sgbd}\enspace{\scriptsize\color{black!60}\tipo}\enspace\uso};
      \draw[seta] (tool.east) -- (n\i.west);
    }
    \draw[seta] (csv.east) -- (tool.west);
  \end{tikzpicture}\par
  \smallskip
  {\footnotesize Cada dado do e-commerce no SGBD mais adequado (Figura~1 do relatório).\par}
\end{frame}

% Slide 4: objetivos (relatório §1.2)
\begin{frame}{Objetivos}
  \begin{painel}{Objetivo geral}{}
    Desenvolver a ferramenta \textit{Polyglot Import CSV}, que importa CSV para
    múltiplos SGBDs: PostgreSQL, Redis, MongoDB, Cassandra ou Neo4j.
  \end{painel}
  \smallskip
  \begin{painel}{Objetivos específicos}{}
    \small
    \begin{enumerate}
      \item \destaque{Configuração} em JSON (texto estruturado): filtros, chave primária e
            nome da coluna.
      \item \destaque{Validação prévia}: um erro na configuração ou nos dados impede
            qualquer gravação.
      \item \destaque{Avaliação} do desempenho com diferentes volumes e modelos de dados.
      \item \destaque{Acesso} por linha de comando (CLI) e por interface gráfica (GUI).
    \end{enumerate}
  \end{painel}
\end{frame}
